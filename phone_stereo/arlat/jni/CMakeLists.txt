# arloop: libarloop.so for the arlat app (Android, -DANDROID_ABI=arm64-v8a) or arloop_replay on the PC.
cmake_minimum_required(VERSION 3.10)
project(arloop CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
if (NOT CMAKE_BUILD_TYPE)
  set(CMAKE_BUILD_TYPE Release)
endif ()

if (ANDROID)
  set(DEPS "$ENV{HOME}/scanner/android")
  set(OpenCV_DIR "${DEPS}/OpenCV-android-sdk/sdk/native/jni")
  set(Ceres_DIR "${DEPS}/ceres-install/lib/cmake/Ceres")
  find_package(OpenCV 4 REQUIRED COMPONENTS core imgproc features2d calib3d objdetect)
  find_package(Ceres REQUIRED)
  add_library(arloop SHARED arloop.cpp arloop_jni.cpp)
  target_include_directories(arloop PRIVATE ${DEPS}/include/eigen3 ${OpenCV_INCLUDE_DIRS})
  target_link_libraries(arloop PRIVATE ${OpenCV_LIBS} Ceres::ceres log)
  target_compile_options(arloop PRIVATE -O3 -Wno-deprecated-declarations)
  target_link_options(arloop PRIVATE -Wl,--gc-sections)
else ()
  list(APPEND CMAKE_PREFIX_PATH "$ENV{HOME}/scanner/local")
  find_package(Eigen3 REQUIRED)
  find_package(OpenCV 4 REQUIRED COMPONENTS core imgproc features2d calib3d aruco)
  find_package(Ceres REQUIRED)
  add_executable(arloop_replay arloop.cpp arloop_replay.cpp)
  target_include_directories(arloop_replay PRIVATE ${EIGEN3_INCLUDE_DIR} ${OpenCV_INCLUDE_DIRS})
  target_link_libraries(arloop_replay PRIVATE ${OpenCV_LIBS} Ceres::ceres pthread)
  target_compile_options(arloop_replay PRIVATE -O2 -Wall)
  set_target_properties(arloop_replay PROPERTIES BUILD_RPATH "$ENV{HOME}/scanner/local/lib")
endif ()
